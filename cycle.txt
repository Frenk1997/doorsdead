# ============================================================
#  Цикл фарма смертей в DOORS
#  Строки с # — комментарии. Команды описаны в README.md
#  Перед запуском запомни точки в окне макроса:
#    @revive      — кнопка «Возродиться» (Revive) на экране смерти
#    @play_again  — кнопка «Играть снова» (Play Again) после смерти
#  (необязательно) @skip — место, куда кликать, чтобы пролистать
#    реплики «Путеводного света» на экране смерти
# ============================================================

focus                      # сделать окно Roblox активным

label start
log Новый забег
wait 2                     # ждём, пока лифт откроется (подстрой под себя)

# --- 1. Вышли из лифта, взяли ключ, открыли дверь ---
run route.txt              # маршрут, записанный кнопкой F7

# --- 2. Смерть от Eyes: просто ждём экран смерти ---
label wait_death1
wait_pixel @revive 90 no_revive    # ждём кнопку «Возродиться» до 90 с
count
wait 0.5
click @revive
log Возродились

# --- 3. После возрождения быстро умираем снова ---
wait 3
# Если после возрождения нужно повернуться к Eyes или пройти пару шагов,
# запиши второй маршрут в after_revive.txt и раскомментируй строку:
# run after_revive.txt
wait_pixel @play_again 90 no_play_again
count
wait 0.5
click @play_again

# --- 4. Ждём загрузку нового забега и повторяем ---
wait 15
goto start

# ------------------------------------------------------------
label no_revive
log Кнопка «Возродиться» не появилась — пробуем сразу «Играть снова»
if_pixel @play_again play_again_now
goto start

label no_play_again
log Кнопка «Играть снова» не появилась — пробуем ещё раз
wait_pixel @play_again 60 start

label play_again_now
count
click @play_again
wait 15
goto start
